24/// Whether one category participates for this configuration: toggles for the
25/// regular categories, the (inverted) attachments flag for ProjectAttachments.
26pub fn is_category_enabled(desc: &CategoryDescriptor, filter: &FilterConfig) -> bool {
27    match desc.id {
28        CategoryId::ProjectAttachments => !filter.exclude_attachments,
29        _ => filter.sync_artifacts.is_enabled(desc.id),
30    }
31}

46    match desc.dest {

52/// True when a file extension is excluded for this category.
53fn extension_excluded(desc: &CategoryDescriptor, path: &Path) -> bool {
54    path.extension()
55        .and_then(|e| e.to_str())
56        .map(|ext| {
57            desc.exclude_extensions
58                .iter()
59                .any(|x| ext.eq_ignore_ascii_case(x))
60        })
61        .unwrap_or(false)
62}

117/// Outcome of one artifact push or pull across all enabled categories.
118#[derive(Debug, Clone, Default)]
119pub struct ArtifactReport {
120    pub counts: Vec<CategoryCounts>,
121    /// Every file written or removed, in the order it happened.
122    pub changes: Vec<ArtifactChange>,
123}

125impl ArtifactReport {
126    pub fn total_added(&self) -> usize {
127        self.counts.iter().map(|c| c.added).sum()
128    }
129    pub fn total_modified(&self) -> usize {
130        self.counts.iter().map(|c| c.modified).sum()
131    }
132    pub fn total_unchanged(&self) -> usize {
133        self.counts.iter().map(|c| c.unchanged).sum()
134    }
135    pub fn total_deleted(&self) -> usize {
136        self.counts.iter().map(|c| c.deleted).sum()
137    }
138
139    fn record(&mut self, category: CategoryId, kind: ArtifactChangeKind, path: &Path) {
140        self.changes.push(ArtifactChange {
141            category,
142            kind,
143            path: path.to_path_buf(),
144        });
145    }
146
147    /// True when nothing was copied, merged, or even inspected.
148    #[allow(dead_code)] // used via the library target; the bin compiles this module separately
149    pub fn is_empty(&self) -> bool {
150        self.counts.is_empty()
151            || self
152                .counts
153                .iter()
154                .all(|c| c.added + c.modified + c.unchanged + c.skipped == 0)
155    }
156}

158/// One collected artifact file: its absolute source under `~/.claude` and its
159/// destination path relative to the category's repo subdirectory.
160struct CollectedFile {
161    abs: PathBuf,
162    rel: PathBuf,
163}
164
165/// Enumerate a category's files on disk. Missing sources yield an empty list;
166/// denied paths and oversized files are skipped (the latter counted).
167///
168/// An oversized file still exists on this machine, so its category-relative
169/// path goes into `held_back`: deletion mirroring must not read "not pushed
170/// this time" as "deleted here" and remove every machine's copy.
171fn collect(
172    desc: &CategoryDescriptor,
173    claude_dir: &Path,
174    filter: &FilterConfig,
175    skipped: &mut usize,
176    held_back: &mut Vec<PathBuf>,
177) -> Result<Vec<CollectedFile>> {
178    let max_file_size = filter.max_file_size_bytes;
179    let mut files = Vec::new();
180
181    match desc.source {
182        SourceSpec::Files(list) => {
183            for entry in list {
184                let claude_rel = Path::new(entry);
185                if is_denied(claude_rel) {
186                    *skipped += 1;
187                    continue;
188                }
189                let abs = claude_dir.join(entry);
190                if !abs.is_file() {
191                    continue;
192                }
193                let rel = claude_rel
194                    .file_name()
195                    .map(PathBuf::from)
196                    .unwrap_or_else(|| claude_rel.to_path_buf());
197                files.push(CollectedFile { abs, rel });
198            }
199        }
200        SourceSpec::Dir(dir) => {
201            let base = claude_dir.join(dir);
202            if !base.is_dir() {
203                return Ok(files);
204            }
205            // Resolved once per project: in name-only mode it reads a transcript.
206            let mut project_names: std::collections::HashMap<String, String> =
207                std::collections::HashMap::new();
208            for entry in walkdir::WalkDir::new(&base)
209                .follow_links(false)
210                .into_iter()
211                .filter_map(|e| e.ok())
212            {
213                if !entry.file_type().is_file() {
214                    continue;
215                }
216                let abs = entry.path();
217                // Deny rules match against the ~/.claude-relative path so a
218                // component like `cache/` is caught wherever it appears.
219                let claude_rel = abs.strip_prefix(claude_dir).unwrap_or(abs);
220                if is_denied(claude_rel) {
221                    *skipped += 1;
222                    continue;
223                }
224                if extension_excluded(desc, abs) {
225                    continue;
226                }
227                let mut rel = abs.strip_prefix(&base).unwrap_or(abs).to_path_buf();
228                // Attachments take the project's repo-side directory name,
229                // mirroring session layout.
230                if desc.dest == DestRoot::SessionTree {
231                    let mut parts = rel.components();
232                    let Some(encoded) = parts.next().and_then(|c| c.as_os_str().to_str()) else {
233                        *skipped += 1;
234                        continue;
235                    };
236                    let project = project_names
237                        .entry(encoded.to_string())
238                        .or_insert_with(|| {
239                            crate::project_map::repo_dir_name_in(filter, &base, encoded)
240                        })
241                        .clone();
242                    rel = Path::new(&project).join(parts.as_path());
243                }
244                if entry.metadata().map(|m| m.len()).unwrap_or(0) > max_file_size {
245                    log::warn!("Skipping {} (exceeds max_file_size_bytes)", abs.display());
246                    *skipped += 1;
247                    held_back.push(rel);
248                    continue;
249                }
250                files.push(CollectedFile {
251                    abs: abs.to_path_buf(),
252                    rel,
253                });
254            }
255        }
256    }
257
258    Ok(files)
259}

261/// Write `content` to `path` via a same-directory temp file + atomic rename,
262/// so a reader (or a crash) never sees a half-written file. `mode_source` is
263/// the file whose executable bit the result adopts.
264#[cfg_attr(not(unix), allow(unused_variables))]
265fn write_atomic(path: &Path, content: &[u8], mode_source: &Path) -> Result<()> {
266    let parent = path
267        .parent()
268        .with_context(|| format!("No parent directory for {}", path.display()))?;
269    fs::create_dir_all(parent)?;
270    let tmp = tempfile::NamedTempFile::new_in(parent)?;
271    fs::write(tmp.path(), content)?;
272    #[cfg(unix)]
273    {
274        if let Some(mode) = get_mode_for_copy(path, mode_source) {
275            set_mode(tmp.path(), mode);
276        }
277    }
278    tmp.persist(path)
279        .with_context(|| format!("Failed to persist {}", path.display()))?;
280    Ok(())
281}

283/// The mode a copy of `mode_source` should have at `destination`: what the
284/// destination already has, plus the owner's executable bit when the source is
285/// executable. Granting only: a repository written before this bit was synced
286/// holds every file non-executable, and a pull from it must not disarm the
287/// scripts on this machine.
288#[cfg(unix)]
289fn get_mode_for_copy(destination: &Path, mode_source: &Path) -> Option<u32> {
290    use std::os::unix::fs::PermissionsExt;
291
292    let source_mode = fs::metadata(mode_source).ok()?.permissions().mode();
293    let base = match fs::metadata(destination) {
294        Ok(existing) => existing.permissions().mode() & 0o777,
295        Err(_) => 0o600,
296    };
297    if source_mode & 0o111 == 0 {
298        return Some(base);
299    }
300    Some(base | 0o100)
301}

303/// Apply `mode` and report whether the file actually carries it afterwards. A
304/// filesystem without permission bits (exFAT, some CIFS mounts) either refuses
305/// the call or ignores it; either way the sync continues and the file counts as
306/// unchanged, instead of being offered again on every run.
307#[cfg(unix)]
308fn set_mode(path: &Path, mode: u32) -> bool {
309    use std::os::unix::fs::PermissionsExt;
310
311    if let Err(error) = fs::set_permissions(path, fs::Permissions::from_mode(mode)) {
312        log::warn!("Could not set permissions on {}: {error}", path.display());
313        return false;
314    }
315    let applied = fs::metadata(path).map(|m| m.permissions().mode() & 0o777);
316    applied.is_ok_and(|applied| applied == mode)
317}

319/// Whether `path` is missing an executable bit that `mode_source` has. Content
320/// comparison alone never notices a `chmod +x`, which would leave the bit stuck
321/// at whatever it was when the file was first copied.
322#[cfg(unix)]
323fn executable_bit_differs(path: &Path, mode_source: &Path) -> bool {
324    use std::os::unix::fs::PermissionsExt;
325
326    if is_symlink(path) {
327        return false;
328    }
329    let Some(wanted) = get_mode_for_copy(path, mode_source) else {
330        return false;
331    };
332    let Ok(metadata) = fs::metadata(path) else {
333        return false;
334    };
335    metadata.permissions().mode() & 0o777 != wanted
336}
337
338#[cfg(not(unix))]
339fn executable_bit_differs(_path: &Path, _mode_source: &Path) -> bool {
340    false
341}
342
343/// A chmod follows symlinks, so it would reach a file outside `~/.claude` that
344/// a write never touches: `write_atomic` replaces the link itself.
345#[cfg(unix)]
346fn is_symlink(path: &Path) -> bool {
347    fs::symlink_metadata(path).is_ok_and(|metadata| metadata.file_type().is_symlink())
348}
349
350/// Bring `path`'s executable bit in line with `mode_source`, reporting whether
351/// the file changed.
352#[cfg(unix)]
353fn align_executable_bit(path: &Path, mode_source: &Path) -> bool {
354    if !executable_bit_differs(path, mode_source) {
355        return false;
356    }
357    let Some(mode) = get_mode_for_copy(path, mode_source) else {
358        return false;
359    };
360    set_mode(path, mode)
361}
362
363#[cfg(not(unix))]
364fn align_executable_bit(_path: &Path, _mode_source: &Path) -> bool {
365    false
366}

368/// Copy every enabled artifact category into `<repo_root>/artifacts/`,
369/// classifying each file Added/Modified/Unchanged by byte comparison.
370/// Prompt history and memory indexes are union-merged into the repo copy
371/// instead of overwritten, config files are path-tokenized, and a file this
372/// machine previously synced and has since deleted is removed from the repo.
373pub fn push_artifacts(
374    claude_dir: &Path,
375    repo_root: &Path,
376    filter: &FilterConfig,
377) -> Result<ArtifactReport> {
378    let mut report = ArtifactReport::default();
379    let tokens = PathTokens::for_claude_dir(claude_dir);
380    let tracked_before = tracked::load(claude_dir, repo_root);
381    let mut tracked_now = TrackedPaths::new();
382
383    for desc in active_categories(filter) {
384        let mut counts = CategoryCounts::new(desc.id);
385
386        let mut held_back = Vec::new();
387        let files = collect(
388            desc,
389            claude_dir,
390            filter,
391            &mut counts.skipped,
392            &mut held_back,
393        )?;
394        let category_root = category_repo_root(desc, repo_root, filter);
395        let mut pushed: TrackedPaths = TrackedPaths::new();
396        // Files kept back for their size are still here: count them as
397        // present so the repo copy (if any) is neither deleted nor forgotten.
398        for rel in held_back {
399            if let Some(rel) = repo_relative(repo_root, &category_root.join(rel)) {
400                pushed.insert(rel);
401            }
402        }
403
404        for file in files {
405            let dest = category_root.join(&file.rel);
406
407            match desc.merge {
408                MergeStrategy::UnionJsonl => {
409                    let local_text = fs::read_to_string(&file.abs).unwrap_or_default();
410                    let existed = dest.is_file();
411                    let repo_text = if existed {
412                        fs::read_to_string(&dest).unwrap_or_default()
413                    } else {
414                        String::new()
415                    };
416                    let (merged, new_lines) = merge_history_lines(&repo_text, &local_text);
417                    counts.merged_entries += new_lines;
418                    if !existed {
419                        write_atomic(&dest, merged.as_bytes(), &file.abs)?;
420                        counts.added += 1;
421                        report.record(desc.id, ArtifactChangeKind::Added, &file.rel);
422                    } else if merged != repo_text {
423                        write_atomic(&dest, merged.as_bytes(), &file.abs)?;
424                        counts.modified += 1;
425                        report.record(desc.id, ArtifactChangeKind::Modified, &file.rel);
426                    } else {
427                        counts.unchanged += 1;
428                    }
429                }
430                MergeStrategy::UnionMemoryIndex | MergeStrategy::RawOverwrite => {
431                    let read_bytes = fs::read(&file.abs).with_context(|| {
432                        format!("Failed to read artifact {}", file.abs.display())
433                    })?;
434                    let mut src_bytes = if desc.tokenize_paths {
435                        tokens.to_repo(&read_bytes)
436                    } else {
437                        read_bytes
438                    };
439                    let existed = dest.is_file();
440                    let unions_index =
441                        desc.merge == MergeStrategy::UnionMemoryIndex && is_memory_index(&file.rel);
442                    if existed && unions_index {
443                        let (merged, new_entries) =
444                            merge_memory_index(&fs::read(&dest)?, &src_bytes);
445                        counts.merged_entries += new_entries;
446                        src_bytes = merged;
447                    }
448
449                    if !existed {
450                        write_atomic(&dest, &src_bytes, &file.abs)?;
451                        counts.added += 1;
452                        report.record(desc.id, ArtifactChangeKind::Added, &file.rel);
453                    } else if fs::read(&dest)? != src_bytes {
454                        write_atomic(&dest, &src_bytes, &file.abs)?;
455                        counts.modified += 1;
456                        report.record(desc.id, ArtifactChangeKind::Modified, &file.rel);
457                    } else {
458                        let realigned = align_executable_bit(&dest, &file.abs);
459                        if realigned {
460                            counts.modified += 1;
461                            report.record(desc.id, ArtifactChangeKind::Modified, &file.rel);
462                        } else {
463                            counts.unchanged += 1;
464                        }
465                    }
466                }
467            }
468
469            if let Some(rel) = repo_relative(repo_root, &dest) {
470                pushed.insert(rel);
471            }
472        }
473
474        if desc.mirror_deletes {
475            if source_is_present(desc, claude_dir) {
476                mark_category_synced(&category_root)?;
477                let removed =
478                    remove_from_repo(&tracked_before, &pushed, &category_root, repo_root)?;
479                counts.deleted += removed.len();
480                for path in &removed {
481                    report.record(desc.id, ArtifactChangeKind::Deleted, path);
482                }
483                tracked_now.extend(pushed);
484            } else {
485                // A category this machine does not have says nothing about
486                // what the others hold: leave the repo copy and the record.
487                log::info!(
488                    "Category {} is not present under {}; its repo copy is left untouched",
489                    desc.name,
490                    claude_dir.display()
491                );
492                tracked_now.extend(tracked_under(&tracked_before, &category_root, repo_root));
493            }
494        }
495
496        report.counts.push(counts);
497    }
498
499    if active_categories(filter).any(|desc| desc.mirror_deletes) {
500        tracked::save(claude_dir, repo_root, tracked_now)?;
501    }
502    Ok(report)
503}

505/// Marker file that keeps a category's repo directory present once its last
506/// real file is deleted.
507///
508/// Git does not track directories, so an emptied category would disappear and
509/// be read as "this repo has no such category", which pull must not delete
510/// for. The marker distinguishes an empty category from an absent one.
511pub const CATEGORY_MARKER: &str = ".synced";
512
513/// Keep the category's repo directory alive across an emptying push.
514fn mark_category_synced(category_root: &Path) -> Result<()> {
515    let marker = category_root.join(CATEGORY_MARKER);
516    if marker.is_file() {
517        return Ok(());
518    }
519    fs::create_dir_all(category_root)?;
520    fs::write(&marker, b"").with_context(|| format!("Failed to write {}", marker.display()))?;
521    Ok(())
522}

524/// A repo path as a `/`-separated string relative to the repository root.
525fn repo_relative(repo_root: &Path, path: &Path) -> Option<String> {
526    let rel = path.strip_prefix(repo_root).ok()?;
527    Some(rel.to_string_lossy().replace('\\', "/"))
528}

530/// Whether this machine has the source a category copies from. A `Files`
531/// category is always "present": its individual files are optional.
532fn source_is_present(desc: &CategoryDescriptor, claude_dir: &Path) -> bool {
533    match desc.source {
534        SourceSpec::Files(_) => true,
535        SourceSpec::Dir(dir) => claude_dir.join(dir).is_dir(),
536    }
537}

539/// The tracked paths that belong to one category's repo directory.
540fn tracked_under(tracked: &TrackedPaths, category_root: &Path, repo_root: &Path) -> Vec<String> {
541    let Some(prefix) = repo_relative(repo_root, category_root) else {
542        return Vec::new();
543    };
544    let prefix = format!("{prefix}/");
545    tracked
546        .iter()
547        .filter(|path| path.starts_with(&prefix))
548        .cloned()
549        .collect()
550}

552/// Delete the repo copies of files this machine synced before and no longer
553/// has. Returns the removed paths, relative to the category root.
554fn remove_from_repo(
555    tracked_before: &TrackedPaths,
556    pushed: &TrackedPaths,
557    category_root: &Path,
558    repo_root: &Path,
559) -> Result<Vec<PathBuf>> {
560    let mut removed = Vec::new();
561    let category_prefix = repo_relative(repo_root, category_root).unwrap_or_default();
562    for gone in tracked_under(tracked_before, category_root, repo_root) {
563        if pushed.contains(&gone) {
564            continue;
565        }
566        let path = repo_root.join(&gone);
567        if path.is_file() {
568            fs::remove_file(&path)
569                .with_context(|| format!("Failed to remove {}", path.display()))?;
570            let category_relative = Path::new(&gone)
571                .strip_prefix(&category_prefix)
572                .map(Path::to_path_buf)
573                .unwrap_or_default();
574            removed.push(category_relative);
575        }
576    }
577    Ok(removed)
578}

592/// Read-only classification of an artifact pull, computed BEFORE any write so
593/// the caller can snapshot the exact set of files that will change.
594#[derive(Debug, Default)]
595pub struct PullPlan {

597    pub overwrites: Vec<PlannedWrite>,
598    /// This machine and the remote both changed the file since they last
599    /// agreed on it.
600    pub changed_on_both_sides: Vec<PlannedWrite>,
601    /// Edited or deleted here since the last sync, while the repo copy stayed
602    /// as it was: left alone for the next push.
603    pub kept_local: Vec<PlannedWrite>,
604    /// Whether a person is at a terminal to settle a file both sides changed.
605    pub can_ask: bool,
606    /// No local file yet: created, and recorded for deletion on undo.
607    pub creates: Vec<PlannedWrite>,
608    /// Union-merge targets whose local file would gain lines.
609    pub unions: Vec<PlannedWrite>,
610    /// Local files whose content already matches but whose executable bit does
611    /// not: a `chmod +x` elsewhere, with nothing to rewrite.
612    pub mode_fixes: Vec<PlannedWrite>,
613    /// Local files this machine synced before that the repo no longer has.
614    pub deletes: Vec<PlannedDelete>,
615    pub unchanged: usize,
616    /// Repo files refused (denied names, unsafe paths).
617    pub skipped: usize,
618    /// Repo files whose project this machine has no destination for, grouped
619    /// by the project directory they came from, so the caller can warn once
620    /// per project instead of once per file.
621    pub unmapped_projects: crate::project_map::SkippedByProject,
622    /// This machine's path tokens, so applying renders repo bytes the same way
623    /// planning compared them.
624    pub tokens: PathTokens,
625    /// Where to record what this machine holds once the plan is applied.
626    pub claude_dir: PathBuf,
627    pub repo_root: PathBuf,
628    /// The configured external merge tool, offered when a file differs.
629    pub merge_tool: String,
630    /// Whether the overwrite prompt starts on the merge tool.
631    pub prefer_merge_tool: bool,
632    /// The repo paths this machine will hold afterwards, for the next pull to
633    /// tell a deletion from a file it never had.
634    pub tracked_after: TrackedPaths,
635    /// Whether any active category mirrors deletions. When none does, the
636    /// record is left untouched rather than emptied.
637    pub tracks_deletions: bool,
638}

640/// One planned local deletion during a pull.
641#[derive(Debug, Clone)]
642pub struct PlannedDelete {
643    pub category: CategoryId,
644    /// Absolute path under `~/.claude` to remove.
645    pub local_path: PathBuf,
646    /// Path relative to the category's root, as reported to the user.
647    pub category_path: PathBuf,
648}
649
650impl PullPlan {
651    /// True when applying the plan would write nothing.
652    pub fn is_empty(&self) -> bool {
653        self.overwrites.is_empty()
654            && self.changed_on_both_sides.is_empty()
655            && self.creates.is_empty()
656            && self.unions.is_empty()
657            && self.mode_fixes.is_empty()
658            && self.deletes.is_empty()
659    }
660
661    /// Existing local files the caller must snapshot before applying
662    /// (overwritten raw files, union-merged files and deletions — a snapshot is
663    /// what makes `undo pull` able to bring a deleted file back).
664    ///
665    /// The record of what this machine last synced is included whenever the
666    /// pull would rewrite it, so undoing a pull restores the files *and* the
667    /// record that describes them.
668    pub fn paths_to_snapshot(&self) -> Vec<PathBuf> {
669        let record = tracked::record_path(&self.claude_dir);
670        self.overwrites
671            .iter()
672            .chain(self.changed_on_both_sides.iter())
673            .chain(self.unions.iter())
674            .map(|w| w.local_path.clone())
675            .chain(self.deletes.iter().map(|d| d.local_path.clone()))
676            .chain(record.is_file().then_some(record))
677            .collect()
678    }
679
680    /// Local paths this pull will create; recording them as a snapshot's
681    /// `deleted_files` makes undo remove them again.
682    pub fn created_paths(&self) -> Vec<String> {
683        let record = tracked::record_path(&self.claude_dir);
684        let record_is_new = !record.is_file();
685        self.creates
686            .iter()
687            .map(|w| w.local_path.clone())
688            .chain(record_is_new.then_some(record))
689            .map(|path| path.to_string_lossy().to_string())
690            .collect()
691    }
692}

694/// Enumerate one category's files as stored in the sync repository, returning
695/// (absolute repo path, path relative to the category subdir). Denied and
696/// unsafe paths are refused here, so nothing below ever sees them.
697fn collect_repo_files(
698    desc: &CategoryDescriptor,
699    repo_root: &Path,
700    filter: &FilterConfig,
701    skipped: &mut usize,
702    report_refusals: bool,
703) -> Vec<(PathBuf, PathBuf)> {
704    let category_root = category_repo_root(desc, repo_root, filter);
705    if !category_root.is_dir() {
706        return Vec::new();
707    }
708
709    let mut files = Vec::new();
710    for entry in walkdir::WalkDir::new(&category_root)
711        .follow_links(false)
712        .into_iter()
713        .filter_map(|e| e.ok())
714    {
715        if !entry.file_type().is_file() {
716            continue;
717        }
718        let abs = entry.path();
719        let rel = abs
720            .strip_prefix(&category_root)
721            .unwrap_or(abs)
722            .to_path_buf();
723        if extension_excluded(desc, &rel) {
724            continue;
725        }
726        // The marker exists to keep the directory, and belongs to no machine.
727        if rel == Path::new(CATEGORY_MARKER) {
728            continue;
729        }
730        if is_unsafe_rel_path(&rel) || is_denied(&rel) {
731            if report_refusals {
732                log::warn!(
733                    "Refusing denied/unsafe artifact from sync repo: {}",
734                    abs.display()
735                );
736            }
737            *skipped += 1;
738            continue;
739        }
740        files.push((abs.to_path_buf(), rel));
741    }
742    files
743}

745/// Why a repo file has no local destination. The caller reports the two cases
746/// differently: an unmapped project is one warning for all of its files, an
747/// unrestorable path is a refusal worth its own line.
748enum SkipReason {
749    /// This machine has no directory for the named sync-repo project.
750    UnmappedProject(String),
751    /// Nothing in the category can receive this path.
752    NotRestorable,
753}
754
755/// Map a category-relative repo path back to its absolute local destination
756/// under `~/.claude`, or the reason it has none.
757fn local_destination(
758    desc: &CategoryDescriptor,
759    claude_dir: &Path,
760    rel: &Path,
761    filter: &FilterConfig,
762) -> Result<PathBuf, SkipReason> {
763    match desc.source {
764        // File lists are stored flat in the repo; restore to the listed
765        // location whose file name matches. An unlisted name has NO valid
766        // destination — the allowlist must hold on pull as well as push, or a
767        // poisoned repo could plant arbitrary files at the top of ~/.claude.
768        SourceSpec::Files(list) => list
769            .iter()
770            .find(|entry| Path::new(entry).file_name() == rel.file_name())
771            .map(|entry| claude_dir.join(entry))
772            .ok_or(SkipReason::NotRestorable),
773        SourceSpec::Dir(dir) => {
774            if desc.dest == DestRoot::SessionTree {
775                // Repo path is <project-dir>/<rest>; resolve the leading
776                // component the way session pull does.
777                let split = crate::project_map::split_project_path(rel);
778                let (project, inside_project) = split.ok_or(SkipReason::NotRestorable)?;
779                let projects_dir = claude_dir.join(dir);
780                let local_project =
781                    crate::project_map::local_project_dir(filter, &projects_dir, project)
782                        .ok_or_else(|| SkipReason::UnmappedProject(project.to_string()))?;
783                return Ok(local_project.join(inside_project));
784            }
785            Ok(claude_dir.join(dir).join(rel))
786        }
787    }
788}

790/// The bytes a repo file becomes on this machine before any merge: rendered
791/// back from tokens for config categories, verbatim otherwise.
792fn machine_bytes(
793    desc: &CategoryDescriptor,
794    tokens: &PathTokens,
795    repo_path: &Path,
796) -> Result<Vec<u8>> {
797    let bytes = fs::read(repo_path)
798        .with_context(|| format!("Failed to read artifact {}", repo_path.display()))?;
799    if desc.tokenize_paths {
800        return Ok(tokens.to_machine(&bytes));
801    }
802    Ok(bytes)
803}

805/// Remember, per artifact, the version this machine and the repo now agree on.
806/// A path they disagree on keeps the hash they last agreed on.
807pub fn record_synced_hashes(
808    claude_dir: &Path,
809    repo_root: &Path,
810    filter: &FilterConfig,
811) -> Result<()> {
812    let tokens = PathTokens::for_claude_dir(claude_dir);
813    let previous = tracked::load_synced_hashes(claude_dir, repo_root);
814    let mut agreed = SyncedHashes::new();
815    let mut refused = 0;
816
817    for desc in active_categories(filter) {
818        for (repo_path, rel) in collect_repo_files(desc, repo_root, filter, &mut refused, false) {
819            let union_merged = desc.merge == MergeStrategy::UnionJsonl
820                || (desc.merge == MergeStrategy::UnionMemoryIndex && is_memory_index(&rel));
821            if union_merged {
822                continue;
823            }
824            let Some(key) = repo_relative(repo_root, &repo_path) else {
825                continue;
826            };
827            let Ok(local_path) = local_destination(desc, claude_dir, &rel, filter) else {
828                continue;
829            };
830            let agreed_hash = read_agreed_hash(desc, &tokens, &local_path, &repo_path)?;
831            let last_agreed_hash = previous.get(&key).cloned();
832            if let Some(hash) = agreed_hash.or(last_agreed_hash) {
833                agreed.insert(key, hash);
834            }
835        }
836    }
837
838    tracked::save_synced_hashes(claude_dir, repo_root, agreed)
839}
840
841fn read_agreed_hash(
842    desc: &CategoryDescriptor,
843    tokens: &PathTokens,
844    local_path: &Path,
845    repo_path: &Path,
846) -> Result<Option<String>> {
847    let Ok(local_metadata) = fs::metadata(local_path) else {
848        return Ok(None);
849    };
850    let repo_metadata = fs::metadata(repo_path)?;
851    let sizes_differ = local_metadata.len() != repo_metadata.len();
852    if sizes_differ && !desc.tokenize_paths {
853        return Ok(None);
854    }
855
856    let local_bytes = fs::read(local_path)?;
857    let machine_repo_bytes = machine_bytes(desc, tokens, repo_path)?;
858    let both_hold_it = local_bytes == machine_repo_bytes;
859    Ok(both_hold_it.then(|| tracked::hash_content(&local_bytes)))
860}

870/// Classify what a pull would write, without writing. For raw categories only
871/// the side that changed since this machine last synced wins: remote changes
872/// are written, local edits and deletions stay for the push, and a file both
873/// sides changed is settled when applying. Without a record of the last sync,
874/// remote (repo) bytes win. Union targets are compared against local ∪ remote;
875/// a file this machine synced before and the repo no longer has is a deletion.
876pub fn plan_pull(claude_dir: &Path, repo_root: &Path, filter: &FilterConfig) -> Result<PullPlan> {
877    let mut plan = PullPlan {
878        tokens: PathTokens::for_claude_dir(claude_dir),
879        claude_dir: claude_dir.to_path_buf(),
880        repo_root: repo_root.to_path_buf(),
881        merge_tool: filter.merge_tool.clone(),
882        prefer_merge_tool: filter.prefer_merge_tool,
883        tracks_deletions: active_categories(filter).any(|desc| desc.mirror_deletes),
884        ..Default::default()
885    };
886    let tracked_before = tracked::load(claude_dir, repo_root);
887    let synced_hashes = tracked::load_synced_hashes(claude_dir, repo_root);
888
889    for desc in active_categories(filter) {
890        let category_root = category_repo_root(desc, repo_root, filter);
891        let mut present: TrackedPaths = TrackedPaths::new();
892
893        for (repo_path, rel) in collect_repo_files(desc, repo_root, filter, &mut plan.skipped, true)
894        {
895            let local_path = match local_destination(desc, claude_dir, &rel, filter) {
896                Ok(local_path) => local_path,
897                Err(SkipReason::UnmappedProject(project)) => {
898                    plan.skipped += 1;
899                    plan.unmapped_projects
900                        .entry(project)
901                        .or_default()
902                        .push(repo_path);
903                    continue;
904                }
905                Err(SkipReason::NotRestorable) => {
906                    plan.skipped += 1;
907                    log::warn!(
908                        "Skipping {} (not a file the {} category restores)",
909                        repo_path.display(),
910                        desc.name
911                    );
912                    continue;
913                }
914            };
915            if desc.mirror_deletes {
916                if let Some(tracked_path) = repo_relative(repo_root, &repo_path) {
917                    present.insert(tracked_path);
918                }
919            }
920            let write = PlannedWrite {
921                category: desc.id,
922                local_path: local_path.clone(),
923                repo_path: repo_path.clone(),
924                category_path: rel.clone(),
925            };
926
927            let synced_key = repo_relative(repo_root, &repo_path).unwrap_or_default();
928            let last_synced_hash = synced_hashes.get(&synced_key);
929            let synced_here_before = tracked_before.contains(&synced_key);
930            let may_be_deleted_here = desc.mirror_deletes
931                && last_synced_hash.is_some()
932                && !local_path.is_file()
933                && source_is_present(desc, claude_dir);
934            if may_be_deleted_here {
935                let machine_repo_bytes = machine_bytes(desc, &plan.tokens, &repo_path)?;
936                let repo_hash = tracked::hash_content(&machine_repo_bytes);
937                let repo_unchanged_since_sync = last_synced_hash == Some(&repo_hash);
938                if repo_unchanged_since_sync {
939                    plan.kept_local.push(write);
940                    continue;
941                }
942            }
943
944            match desc.merge {
945                MergeStrategy::UnionJsonl => {
946                    if !local_path.is_file() {
947                        plan.creates.push(write);
948                        continue;
949                    }
950                    let local_text = fs::read_to_string(&local_path).unwrap_or_default();
951                    let repo_text = fs::read_to_string(&repo_path).unwrap_or_default();
952                    let (merged, _) = merge_history_lines(&local_text, &repo_text);
953                    if merged != local_text {
954                        plan.unions.push(write);
955                    } else {
956                        plan.unchanged += 1;
957                    }
958                }
959                MergeStrategy::UnionMemoryIndex if is_memory_index(&rel) => {
960                    if !local_path.is_file() {
961                        plan.creates.push(write);
962                        continue;
963                    }
964                    let local_bytes = fs::read(&local_path)?;
965                    let repo_bytes = machine_bytes(desc, &plan.tokens, &repo_path)?;
966                    let (merged, _) = merge_memory_index(&local_bytes, &repo_bytes);
967                    if merged != local_bytes {
968                        plan.unions.push(write);
969                    } else {
970                        plan.unchanged += 1;
971                    }
972                }
973                MergeStrategy::UnionMemoryIndex | MergeStrategy::RawOverwrite => {
974                    if !local_path.is_file() {
975                        plan.creates.push(write);
976                        continue;
977                    }
978                    let local_bytes = fs::read(&local_path)?;
979                    let machine_repo_bytes = machine_bytes(desc, &plan.tokens, &repo_path)?;
980                    if local_bytes != machine_repo_bytes {
981                        let repo_hash = tracked::hash_content(&machine_repo_bytes);
982                        let local_hash = tracked::hash_content(&local_bytes);
983                        let (repo_changed, local_changed) = match last_synced_hash {
984                            Some(base_hash) => (repo_hash != *base_hash, local_hash != *base_hash),
985                            None if synced_here_before => (true, true),
986                            None => (true, false),
987                        };
988                        if !repo_changed {
989                            plan.kept_local.push(write);
990                        } else if !local_changed {
991                            plan.overwrites.push(write);
992                        } else {
993                            plan.changed_on_both_sides.push(write);
994                        }
995                    } else {
996                        let missing_executable_bit =
997                            executable_bit_differs(&local_path, &repo_path);
998                        if missing_executable_bit {
999                            plan.mode_fixes.push(write);
1000                        } else {
1001                            plan.unchanged += 1;
1002                        }
1003                    }
1004                }
1005            }
1006        }
1007
1008        if !desc.mirror_deletes {
1009            continue;
1010        }
1011
1012        // Mirror of the push-side guard: a category the repo does not have —
1013        // an older or rewound branch — is not "everything was deleted".
1014        if !category_root.is_dir() {
1015            log::info!(
1016                "Category {} is not present in {}; local files are left alone",
1017                desc.name,
1018                repo_root.display()
1019            );
1020            plan.tracked_after
1021                .extend(tracked_under(&tracked_before, &category_root, repo_root));
1022            continue;
1023        }
1024
1025        for gone in tracked_under(&tracked_before, &category_root, repo_root) {
1026            if present.contains(&gone) {
1027                continue;
1028            }
1029            let rel = Path::new(&gone)
1030                .strip_prefix(repo_relative(repo_root, &category_root).unwrap_or_default())
1031                .map(Path::to_path_buf)
1032                .unwrap_or_default();
1033            // The same refusal `collect_repo_files` applies: a record must
1034            // never point a deletion outside its category.
1035            if is_unsafe_rel_path(&rel) || is_denied(&rel) {
1036                log::warn!("Refusing denied/unsafe tracked path: {gone}");
1037                plan.skipped += 1;
1038                continue;
1039            }
1040            let Ok(local_path) = local_destination(desc, claude_dir, &rel, filter) else {
1041                continue;
1042            };
1043            if !local_path.is_file() {
1044                continue;
1045            }
1046            let edited_here_since_sync = match synced_hashes.get(&gone) {
1047                Some(base_hash) => {
1048                    let local_bytes = fs::read(&local_path)?;
1049                    let local_hash = tracked::hash_content(&local_bytes);
1050                    local_hash != *base_hash
1051                }
1052                None => false,
1053            };
1054            if edited_here_since_sync {
1055                log::warn!(
1056                    "{} was deleted on another machine but edited here since; kept, \
1057                     the next push brings it back",
1058                    local_path.display()
1059                );
1060            } else {
1061                plan.deletes.push(PlannedDelete {
1062                    category: desc.id,
1063                    local_path,
1064                    category_path: rel,
1065                });
1066            }
1067        }
1068        plan.tracked_after.extend(present);
1069    }
1070
1071    Ok(plan)
1072}

1074/// Apply a pull plan: create missing files, overwrite the ones only the remote
1075/// changed, settle the ones both sides changed (later dates, else ask when
1076/// `plan.can_ask`, else keep local), union-merge prompt history and memory
1077/// indexes, and remove files the repo no longer has. Under `interactive` in a
1078/// terminal, each overwrite asks for per-file confirmation — with the
1079/// configured merge tool as an option — and each deletion asks for
1080/// confirmation; declined files count as skipped.
1081pub fn apply_pull(plan: &PullPlan, interactive: bool) -> Result<ArtifactReport> {
1082    use std::collections::HashMap;
1083
1084    let mut by_category: HashMap<CategoryId, CategoryCounts> = HashMap::new();
1085    fn counts_for(
1086        map: &mut HashMap<CategoryId, CategoryCounts>,
1087        id: CategoryId,
1088    ) -> &mut CategoryCounts {
1089        map.entry(id)
1090            .or_insert_with(move || CategoryCounts::new(id))
1091    }
1092
1093    let mut report = ArtifactReport::default();
1094    let prompt_overwrites = interactive && plan.can_ask;
1095
1096    for write in &plan.creates {
1097        let bytes = machine_bytes(descriptor(write.category), &plan.tokens, &write.repo_path)?;
1098        write_atomic(&write.local_path, &bytes, &write.repo_path)?;
1099        counts_for(&mut by_category, write.category).added += 1;
1100        report.record(
1101            write.category,
1102            ArtifactChangeKind::Added,
1103            &write.category_path,
1104        );
1105    }
1106
1107    for write in &plan.overwrites {
1108        let bytes = machine_bytes(descriptor(write.category), &plan.tokens, &write.repo_path)?;
1109        let bytes = if prompt_overwrites {
1110            let question = format!("'{}' differs:", write.category_path.display());
1111            let resolved = crate::merge_tool::resolve_overwrite(
1112                &plan.merge_tool,
1113                plan.prefer_merge_tool,
1114                &question,
1115                &write.local_path,
1116                &bytes,
1117            )?;
1118            match resolved {
1119                Some(resolved) => resolved,
1120                None => {
1121                    counts_for(&mut by_category, write.category).skipped += 1;
1122                    continue;
1123                }
1124            }
1125        } else {
1126            bytes
1127        };
1128        write_atomic(&write.local_path, &bytes, &write.repo_path)?;
1129        counts_for(&mut by_category, write.category).modified += 1;
1130        report.record(
1131            write.category,
1132            ArtifactChangeKind::Modified,
1133            &write.category_path,
1134        );
1135    }
1136
1137    for write in &plan.changed_on_both_sides {
1138        let remote_bytes =
1139            machine_bytes(descriptor(write.category), &plan.tokens, &write.repo_path)?;
1140        let local_bytes = fs::read(&write.local_path)
1141            .with_context(|| format!("Failed to read {}", write.local_path.display()))?;
1142        let later_dates_merged = keep_later_timestamps(&local_bytes, &remote_bytes);
1143        let settled = match later_dates_merged {
1144            Some(merged) => Some(merged),
1145            None if plan.can_ask => {
1146                let question =
1147                    format!("'{}' changed on both sides:", write.category_path.display());
1148                crate::merge_tool::resolve_overwrite(
1149                    &plan.merge_tool,
1150                    plan.prefer_merge_tool,
1151                    &question,
1152                    &write.local_path,
1153                    &remote_bytes,
1154                )?
1155            }
1156            None => {
1157                log::warn!(
1158                    "{} changed here and on another machine; kept this machine's version, \
1159                     the next push sends it",
1160                    write.local_path.display()
1161                );
1162                None
1163            }
1164        };
1165        let Some(bytes) = settled else {
1166            counts_for(&mut by_category, write.category).skipped += 1;
1167            continue;
1168        };
1169        let local_already_holds_it = bytes == local_bytes;
1170        if local_already_holds_it {
1171            continue;
1172        }
1173        write_atomic(&write.local_path, &bytes, &write.repo_path)?;
1174        counts_for(&mut by_category, write.category).modified += 1;
1175        report.record(
1176            write.category,
1177            ArtifactChangeKind::Modified,
1178            &write.category_path,
1179        );
1180    }
1181
1182    for write in &plan.unions {
1183        let desc = descriptor(write.category);
1184        let repo_bytes = machine_bytes(desc, &plan.tokens, &write.repo_path)?;
1185        let local_bytes = fs::read(&write.local_path).unwrap_or_default();
1186        let (merged, new_entries) = match desc.merge {
1187            MergeStrategy::UnionMemoryIndex => merge_memory_index(&local_bytes, &repo_bytes),
1188            _ => {
1189                let (text, lines) = merge_history_lines(
1190                    &String::from_utf8_lossy(&local_bytes),
1191                    &String::from_utf8_lossy(&repo_bytes),
1192                );
1193                (text.into_bytes(), lines)
1194            }
1195        };
1196        write_atomic(&write.local_path, &merged, &write.repo_path)?;
1197        let counts = counts_for(&mut by_category, write.category);
1198        counts.modified += 1;
1199        counts.merged_entries += new_entries;
1200        report.record(
1201            write.category,
1202            ArtifactChangeKind::Modified,
1203            &write.category_path,
1204        );
1205    }
1206
1207    for write in &plan.mode_fixes {
1208        if prompt_overwrites && !confirm_executable(&write.local_path) {
1209            counts_for(&mut by_category, write.category).skipped += 1;
1210            continue;
1211        }
1212        let realigned = align_executable_bit(&write.local_path, &write.repo_path);
1213        if realigned {
1214            counts_for(&mut by_category, write.category).modified += 1;
1215            report.record(
1216                write.category,
1217                ArtifactChangeKind::Modified,
1218                &write.category_path,
1219            );
1220        }
1221    }
1222
1223    for delete in &plan.deletes {
1224        if prompt_overwrites && !confirm_deletion(&delete.local_path) {
1225            counts_for(&mut by_category, delete.category).skipped += 1;
1226            continue;
1227        }
1228        if delete.local_path.is_file() {
1229            fs::remove_file(&delete.local_path)
1230                .with_context(|| format!("Failed to remove {}", delete.local_path.display()))?;
1231        }
1232        counts_for(&mut by_category, delete.category).deleted += 1;
1233        report.record(
1234            delete.category,
1235            ArtifactChangeKind::Deleted,
1236            &delete.category_path,
1237        );
1238    }
1239
1240    if plan.tracks_deletions {
1241        tracked::save(
1242            &plan.claude_dir,
1243            &plan.repo_root,
1244            plan.tracked_after.clone(),
1245        )?;
1246    }
1247
1248    report.counts = by_category.into_values().collect();
1249    report.counts.sort_by_key(|c| c.category as usize);
1250    Ok(report)
1251}

1253/// Ask before making a local file runnable, since a hook runs on its own.

1302const IGNORE_BLOCK_START: &str = "# >>> claude-code-sync managed block — do not edit inside";
1303const IGNORE_BLOCK_END: &str = "# <<< claude-code-sync managed block";
1304
1305/// Build the full managed block for one backend.
1306fn ignore_block(backend: Backend) -> String {
1307    let mut block = String::new();
1308    block.push_str(IGNORE_BLOCK_START);
1309    block.push('\n');
1310    if backend == Backend::Mercurial {
1311        block.push_str("syntax: glob\n");
1312    }
1313    for glob in IGNORE_GLOBS {
1314        block.push_str(glob);
1315        block.push('\n');
1316    }
1317    block.push_str(IGNORE_BLOCK_END);
1318    block.push('\n');
1319    block
1320}

1322/// Write the managed never-sync ignore block into the sync repository's
1323/// ignore file for the given backend. Idempotent; preserves user content
1324/// outside the block. Returns whether the file changed.
1325pub fn ensure_ignore_files(repo_root: &Path, backend: Backend) -> Result<bool> {
1326    let file_name = match backend {
1327        Backend::Git => ".gitignore",
1328        Backend::Mercurial => ".hgignore",
1329    };
1330    let path = repo_root.join(file_name);
1331    let existing = if path.is_file() {
1332        fs::read_to_string(&path)?
1333    } else {
1334        String::new()
1335    };
1336
1337    let block = ignore_block(backend);
1338
1339    let updated = if let (Some(start), Some(end)) = (
1340        existing.find(IGNORE_BLOCK_START),
1341        existing.find(IGNORE_BLOCK_END),
1342    ) {
1343        // Replace the existing block in place.
1344        let end = end + IGNORE_BLOCK_END.len();
1345        // Include the trailing newline of the old block if present.
1346        let end = if existing[end..].starts_with('\n') {
1347            end + 1
1348        } else {
1349            end
1350        };
1351        format!("{}{}{}", &existing[..start], block, &existing[end..])
1352    } else if existing.is_empty() {
1353        block
1354    } else {
1355        let sep = if existing.ends_with('\n') {
1356            "\n"
1357        } else {
1358            "\n\n"
1359        };
1360        format!("{existing}{sep}{block}")
1361    };
1362
1363    if updated == existing {
1364        return Ok(false);
1365    }
1366    fs::write(&path, updated)?;
1367    Ok(true)
1368}
